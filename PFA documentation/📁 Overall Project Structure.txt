    name: inference-service

      group-id: inference-group
      key-deserializer: org.apache.kafka.common.serialization.StringDeserializer

      properties:
        spring.json.trusted.packages: "com.pbads.data.model.event"

  model:
    cache-ttl: 3600 # seconds
    default-threshold: 0.95
  fallback:
    enabled: true
    rule-based-thresholds:
      sleep_hours: {min: 6, max: 9}
      mood_score: {min: 3, max: 5}

9. 🐳 Docker & Orchestration
text
docker-compose.yml
yaml
version: '3.8'

services:
  eureka-server:
    build: ./eureka-server
    ports:
      - "8761:8761"
    networks:
      - pbads-network

  api-gateway:
    build: ./api-gateway
    ports:
      - "8080:8080"
    depends_on:
      - eureka-server
    environment:
      - EUREKA_CLIENT_SERVICEURL_DEFAULTZONE=http://eureka-server:8761/eureka
    networks:
      - pbads-network

  data-service:
    build: ./data-service
    ports:
      - "8081:8081"
    depends_on:
      - eureka-server
      - postgres-db
      - kafka
    environment:
      - SPRING_DATASOURCE_URL=jdbc:postgresql://postgres-db:5432/pbads_data
      - EUREKA_CLIENT_SERVICEURL_DEFAULTZONE=http://eureka-server:8761/eureka
    networks:
      - pbads-network

  # ... other services similar

  postgres-db:
    image: postgres:14
    environment:
      POSTGRES_DB: pbads_data
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: password
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
    networks:
      - pbads-network

  kafka:
    image: confluentinc/cp-kafka:latest
    depends_on:
      - zookeeper
    environment:
      KAFKA_BROKER_ID: 1
      KAFKA_ZOOKEEPER_CONNECT: zookeeper:2181
      KAFKA_ADVERTISED_LISTENERS: PLAINTEXT://kafka:9092
      KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 1
    ports:
      - "9092:9092"
    networks:
      - pbads-network

  zookeeper:
    image: confluentinc/cp-zookeeper:latest
    environment:
      ZOOKEEPER_CLIENT_PORT: 2181
      ZOOKEEPER_TICK_TIME: 2000
    ports:
      - "2181:2181"
    networks:
      - pbads-network

networks:
  pbads-network:
    driver: bridge

volumes:
  postgres_data:
🎯 Key Configuration Points:
Eureka Integration: Each service needs @EnableEurekaClient annotation

Feign Clients: For service-to-service communication

Kafka Topics: Define topics in a shared configuration:

data.logged.events

model.updated.events

anomaly.detected.events

Service Ports:

Eureka: 8761

Gateway: 8080

Data Service: 8081

Model Service: 8082

Inference Service: 8083

Alert Service: 8084

Database Per Service: Each service should have its own database schema

This structure follows Domain-Driven Design principles with clear separation of concerns, making it easy to scale, test, and maintain each microservice independently.



1. Eureka Server (Service Discovery)
2. Config Server (Configuration)
3. API Gateway (Routing + Security)
4. Auth Service (JWT/OAuth2)
5. Data Service (PostgreSQL)
6. Model Service (Python/Java hybrid)
7. Inference Service (Redis cache for models)
8. Alert Service (WebSocket + Email)
9. Monitoring Service (Prometheus + Grafana)
10. Model Registry (MLflow)
11. Message Broker (Kafka)
12. Frontend (React + WebSocket)
