```yaml
version: '3.8'
services:
  # Databases
  postgres-data:
    image: postgres:14
    environment:
      POSTGRES_DB: pbads_data
      POSTGRES_USER: pbads
      POSTGRES_PASSWORD: pbads_password
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
  
  postgres-auth:
    image: postgres:14
    environment:
      POSTGRES_DB: pbads_auth
      POSTGRES_USER: pbads
      POSTGRES_PASSWORD: pbads_password
    ports:
      - "5433:5432"
    volumes:
      - postgres_auth:/var/lib/postgresql/data

  # Kafka
  zookeeper:
    image: confluentinc/cp-zookeeper:latest
    environment:
      ZOOKEEPER_CLIENT_PORT: 2181
      ZOOKEEPER_TICK_TIME: 2000
    ports:
      - "2181:2181"

  kafka:
    image: confluentinc/cp-kafka:latest
    depends_on:
      - zookeeper
    environment:
      KAFKA_BROKER_ID: 1
      KAFKA_ZOOKEEPER_CONNECT: zookeeper:2181
      KAFKA_ADVERTISED_LISTENERS: PLAINTEXT://kafka:9092
      KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 1
    ports:
      - "9092:9092"

  # Redis
  redis:
    image: redis:7-alpine
    ports:
      - "6379:6379"
    command: redis-server --appendonly yes

  # Monitoring Stack
  prometheus:
    image: prom/prometheus:latest
    ports:
      - "9090:9090"
    volumes:
      - ./monitoring-service/prometheus.yml:/etc/prometheus/prometheus.yml

  grafana:
    image: grafana/grafana:latest
    ports:
      - "3000:3000"
    volumes:
      - ./monitoring-service/grafana:/etc/grafana/provisioning

  # MLflow
  mlflow:
    image: ghcr.io/mlflow/mlflow:latest
    ports:
      - "5000:5000"
    environment:
      - MLFLOW_TRACKING_URI=http://mlflow:5000
    volumes:
      - mlflow_artifacts:/mlflow/artifacts

volumes:
  postgres_data:
  postgres_auth:
  mlflow_artifacts:
```

## **🔑 Key Points to Remember:**

1. **Service Communication**:
   - HTTP/REST via Feign clients
   - Async events via Kafka
   - Real-time via WebSocket

2. **Database Strategy**:
   - Each service has its own database/schema
   - Shared user data via auth service

3. **Security**:
   - JWT tokens for authentication
   - API Gateway validates tokens
   - Service-to-service auth via service accounts

4. **Monitoring**:
   - Prometheus for metrics
   - Grafana for dashboards
   - ELK stack for logs (optional)

5. **MLOps**:
   - MLflow for experiment tracking
   - Model registry for versioning
   - Automated retraining pipelines

This architecture is **production-ready** and follows **best practices** for microservices, MLOps, and cloud-native development. Each service is independently deployable, scalable, and maintainable.
